// v2.0.3: Helper to calculate angular difference with proper wrap-around handling
private func angularDifference(_ a: Double, _ b: Double) -> Double {
    let diff = abs(a - b).truncatingRemainder(dividingBy: 360)
    return diff > 180 ? 360 - diff : diff
}

            guard distanceToMid >= 250 && distanceToMid <= 300 else { return false }
            
            let poiBearing = bearingBetween(midPoint, poi.coordinate)
            let normalizedBearing = poiBearing < 0 ? poiBearing + 360 : poiBearing
            let normalizedRouteDir = routeDirection < 0 ? routeDirection + 360 : routeDirection
            
            // Check if POI is roughly perpendicular (within ±30° of perpendicular)
            let perpLeft = (normalizedRouteDir - 90).truncatingRemainder(dividingBy: 360)
            let perpRight = (normalizedRouteDir + 90).truncatingRemainder(dividingBy: 360)
            
            let diffLeft = min(abs(normalizedBearing - perpLeft), 360 - abs(normalizedBearing - perpLeft))
            let diffRight = min(abs(normalizedBearing - perpRight), 360 - abs(normalizedBearing - perpRight))
            
            return min(diffLeft, diffRight) <= 30  // Within 30° of perpendicular
        }
        
        if !perpendicularPOIs.isEmpty {
            print("🗺️ 🔍 [PASS 2] Found \(perpendicularPOIs.count) perpendicular POIs")
            
            // Try the closest perpendicular POI
            let sortedPerp = perpendicularPOIs.sorted {
                distanceBetween($0.coordinate, midPoint) < distanceBetween($1.coordinate, midPoint)
            }
            
            if let detourPOI = sortedPerp.first {
                // Insert detour POI near midpoint of route
                var detourWaypoints = orderedWaypoints
                let insertIndex = min(detourWaypoints.count / 2, detourWaypoints.count - 1)
                detourWaypoints.insert(detourPOI, at: insertIndex)
                
                if let detourDirections = try? await getWalkingDirections(
                    origin: origin,
                    destination: origin,
                    waypoints: detourWaypoints.map { $0.coordinate }
                ) {
                    let detourDuration = detourDirections.legs.reduce(0) { $0 + $1.duration.value }
                    let detourDistance = detourDirections.legs.reduce(0) { $0 + $1.distance.value }
                    let detourMins = detourDuration / 60
                    
                    if detourDuration >= minAcceptable && detourDuration <= maxAcceptable {
                        print("🗺️ 🔍 [PASS 2] ✅ PERPENDICULAR DETOUR: \(durationMin)min → \(detourMins)min (added '\(detourPOI.name)')")
                        
                        // Reorder if Google provided order
                        var orderedDetourWaypoints = detourWaypoints
                        if let order = detourDirections.waypointOrder, order.count == detourWaypoints.count {
                            orderedDetourWaypoints = order.compactMap { idx in
                                idx < detourWaypoints.count ? detourWaypoints[idx] : nil
                            }
                        }
                        
                        let detourRoute = GeneratedRoute(
                            places: orderedDetourWaypoints,
                            polyline: detourDirections.overviewPolyline.points,
                            distanceMeters: detourDistance,
                            durationSeconds: detourDuration,
                            legs: detourDirections.legs
                        )
                        validRoutes.append(detourRoute)
                        routeCapture?.addRoute(detourRoute)
                        return finalizeRouteDedup(detourRoute)
                    } else {
                        print("🗺️ 🔍 [PASS 2] ✗ Detour route outside tolerance: \(detourMins)min")
                    }
                }
            }
        } else {
            print("🗺️ 🔍 [PASS 2] No suitable perpendicular POIs found")
        }
    }
}
