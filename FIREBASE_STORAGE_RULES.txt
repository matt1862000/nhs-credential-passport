service firebase.storage {
  match /b/{bucket}/o {
    // Allow public read access to pre-populated POI database
    match /prepopulated_pois.json {
      allow read: if true;  // Public read access for app downloads
      allow write: if false; // Only admins can write (via console)
    }
    
    // All other files remain secure
    match /{allPaths=**} {
      allow read, write: if false;
    }
  }
}
